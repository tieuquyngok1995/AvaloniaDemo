using System;
using System.Diagnostics;
using System.IO;
using System.Threading.Tasks;
using ConfigGenerator.Interfaces;

namespace ConfigGenerator.Service;

public class JsonServerService : IJsonServerService, IDisposable
{


    private Process? _jsonServerProcess;
    private readonly string _nodeFolder;
    private readonly string _appFolder;
    private readonly string _dbFile;
    private readonly int _port;
    private bool _disposed = false;

    public JsonServerService(string nodeFolder, string appFolder, string dbFileName = "db.json", int port = 7200)
    {
        _nodeFolder = nodeFolder ?? throw new ArgumentNullException(nameof(nodeFolder));
        _appFolder = appFolder ?? throw new ArgumentNullException(nameof(appFolder));
        _dbFile = Path.Combine(appFolder, dbFileName);
        _port = port;

        Debug.WriteLine($"JsonServerService initialized - NodeFolder: {_nodeFolder}, AppFolder: {_appFolder}, Port: {_port}");
    }

    private bool NodeExists()
    {
        string nodeExe = Path.Combine(_nodeFolder, "node.exe");
        string npmCmd = Path.Combine(_nodeFolder, "npm.cmd");
        bool exists = File.Exists(nodeExe) && File.Exists(npmCmd);

        Debug.WriteLine($"Node exists check - NodeExe: {File.Exists(nodeExe)}, NpmCmd: {File.Exists(npmCmd)}");
        return exists;
    }

    private async Task<bool> EnsureJsonServerInstalledAsync()
    {
        try
        {
            string nodeModules = Path.Combine(_appFolder, "node_modules", "json-server");
            if (Directory.Exists(nodeModules))
            {
                Debug.WriteLine("json-server already installed locally");
                return true;
            }

            Debug.WriteLine("Installing json-server locally...");

            var psi = new ProcessStartInfo
            {
                FileName = Path.Combine(_nodeFolder, "npm.cmd"),
                Arguments = "install json-server --prefix . --no-audit --no-fund",
                WorkingDirectory = _appFolder,
                UseShellExecute = false,
                RedirectStandardOutput = true,
                RedirectStandardError = true,
                CreateNoWindow = true
            };

            using var proc = Process.Start(psi);
            if (proc == null)
            {
                Debug.WriteLine("Failed to start npm install process");
                return false;
            }

            var outputTask = proc.StandardOutput.ReadToEndAsync();
            var errorTask = proc.StandardError.ReadToEndAsync();

            await proc.WaitForExitAsync();

            string output = await outputTask;
            string error = await errorTask;

            if (!string.IsNullOrEmpty(output))
                Debug.WriteLine($"[INSTALL OUTPUT] {output}");

            if (proc.ExitCode != 0)
            {
                Debug.WriteLine($"npm install failed with exit code: {proc.ExitCode}");
                if (!string.IsNullOrEmpty(error))
                    Debug.WriteLine($"[INSTALL ERROR] {error}");
                return false;
            }

            Debug.WriteLine("json-server installed successfully");
            return true;
        }
        catch (Exception ex)
        {
            Debug.WriteLine($"Exception during json-server installation: {ex.Message}");
            return false;
        }
    }

    public async Task<bool> StartServerAsync()
    {
        try
        {
            if (_jsonServerProcess != null && !_jsonServerProcess.HasExited)
            {
                Debug.WriteLine("Server is already running");
                return true;
            }

            if (!NodeExists())
            {
                Debug.WriteLine("Node.js not found in specified folder");
                return false;
            }

            if (!File.Exists(_dbFile))
            {
                Debug.WriteLine($"Database file not found: {_dbFile}");
                return false;
            }

            if (!await EnsureJsonServerInstalledAsync())
            {
                Debug.WriteLine("Failed to ensure json-server installation");
                return false;
            }

            var psi = new ProcessStartInfo
            {
                FileName = Path.Combine(_nodeFolder, "npx.cmd"),
                Arguments = $"--yes json-server --watch \"{_dbFile}\" --port {_port} --host localhost",
                WorkingDirectory = _appFolder,
                UseShellExecute = false,
                RedirectStandardOutput = true,
                RedirectStandardError = true,
                CreateNoWindow = true
            };

            Debug.WriteLine($"Starting json-server: {psi.FileName} {psi.Arguments}");
            Debug.WriteLine($"Working Directory: {psi.WorkingDirectory}");

            _jsonServerProcess = Process.Start(psi);

            if (_jsonServerProcess == null)
            {
                Debug.WriteLine("Failed to start json-server process");
                return false;
            }

            Debug.WriteLine($"json-server process started. PID: {_jsonServerProcess.Id}");

            // Setup async output reading
            _jsonServerProcess.OutputDataReceived += (sender, e) =>
            {
                if (!string.IsNullOrEmpty(e.Data))
                    Debug.WriteLine($"[JSON-SERVER] {e.Data}");
            };

            _jsonServerProcess.ErrorDataReceived += (sender, e) =>
            {
                if (!string.IsNullOrEmpty(e.Data))
                    Debug.WriteLine($"[JSON-SERVER ERROR] {e.Data}");
            };

            _jsonServerProcess.EnableRaisingEvents = true;
            _jsonServerProcess.Exited += (sender, e) =>
            {
                Debug.WriteLine($"json-server process exited with code: {_jsonServerProcess?.ExitCode}");
            };

            _jsonServerProcess.BeginOutputReadLine();
            _jsonServerProcess.BeginErrorReadLine();

            // Wait a bit to see if the process starts successfully
            await Task.Delay(2000);

            if (_jsonServerProcess.HasExited)
            {
                Debug.WriteLine($"json-server process exited immediately with code: {_jsonServerProcess.ExitCode}");
                return false;
            }

            Debug.WriteLine($"json-server started successfully on port {_port}");
            return true;
        }
        catch (Exception ex)
        {
            Debug.WriteLine($"Exception when starting json-server: {ex.Message}");
            Debug.WriteLine($"Stack trace: {ex.StackTrace}");
            return false;
        }
    }

    public void StartServer()
    {
        // Synchronous wrapper for backward compatibility
        Task.Run(async () => await StartServerAsync()).Wait();
    }

    public void StopServer()
    {
        try
        {
            if (_jsonServerProcess != null && !_jsonServerProcess.HasExited)
            {
                Debug.WriteLine($"Stopping json-server process (PID: {_jsonServerProcess.Id})");

                _jsonServerProcess.Kill();
                _jsonServerProcess.WaitForExit(5000); // Wait up to 5 seconds

                Debug.WriteLine("json-server process stopped");
            }
        }
        catch (Exception ex)
        {
            Debug.WriteLine($"Exception when stopping json-server: {ex.Message}");
        }
        finally
        {
            _jsonServerProcess?.Dispose();
            _jsonServerProcess = null;
        }
    }

    public bool IsRunning()
    {
        return _jsonServerProcess != null && !_jsonServerProcess.HasExited;
    }

    public string GetServerUrl()
    {
        return $"http://localhost:{_port}";
    }

    public void Dispose()
    {
        if (!_disposed)
        {
            StopServer();
            _disposed = true;
            Debug.WriteLine("JsonServerService disposed");
        }
    }
}
